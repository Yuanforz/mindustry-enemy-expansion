# 防御模块 / Defence modules (our own designs)
#
# Blueprint text format (see src/expansion/Blueprint.java): the first part is the main block, positions are relative to
# it, and the front (+x) faces the enemy. Supply comes in from behind (-x), toward our base.
# An item source or liquid source marks where a supply line ends: on the map it becomes the router that line ends in.
# A part written "?block" goes up only when what it needs is to be had (water for faster fire).
# Lines starting "#!" say what the module is for; the defence planner reads them, the Blueprint reader skips them.
# Walls are not part of a module: the planner lays them as a line across the choke in front (2 tiles thick).
#
# Patterns (reference/research-2026-10-18/schematics-and-sectors.md, P5): turrets in a row with the feed bus behind,
# a router beside each turret, an ammunition store (container and unloader) that fills between waves, and a liquid
# store right behind turrets that fire liquid. Each turret role from reference/turrets-drills-human.md §2.

# ---- 撕裂 Breach: the first line (air and ground) ----

# Two breaches with a store: the line fills a container between waves (160 items), an unloader feeds both turrets
# through a router each. Water, when there is some to spare, comes in between them and cools both (2.5 times the fire).
# Tungsten is 1 an shot and reaches 5 tiles further; beryllium is 2 a shot.
[breach-pair]
#! role line antiAir
#! about two breaches, ammo store, water between them
breach 0 0 0
breach 0 4 0
reinforced-container -3 -3
duct-unloader -2 -1 1
duct-router -2 0 1
duct -2 1 1
duct -2 2 1
duct -2 3 1
duct-router -2 4 1
item-source -4 -3 0 tungsten
?liquid-source -1 2 0 water

# The same without a store, for when there are no containers yet (the first sector): the line ends at the first router.
[breach-pair-direct]
#! role line antiAir
#! about two breaches fed straight from the line
breach 0 0 0
breach 0 4 0
duct-router -2 0 1
duct -2 1 1
duct -2 2 1
duct -2 3 1
duct-router -2 4 1
item-source -2 -1 0 beryllium
?liquid-source -1 2 0 water
